require("dotenv").config(); const { Client, GatewayIntentBits, REST, Routes, SlashCommandBuilder, EmbedBuilder, ActionRowBuilder, ButtonBuilder, ButtonStyle, PermissionFlagsBits, ChannelType } = require("discord.js"); const client = new Client({ intents: [GatewayIntentBits.Guilds] }); // =============================== // CONFIGURAÇÃO DO PRODUTO // =============================== let produto = { nome: "SERAPHIM HUB", preco: "9.50", estoque: 1 }; // =============================== // COMANDOS // =============================== const commands = [ new SlashCommandBuilder() .setName("configurar") .setDescription("Configura o produto") .setDefaultMemberPermissions( PermissionFlagsBits.Administrator ) .addStringOption(option => option .setName("nome") .setDescription("Nome que os clientes verão") .setRequired(true) ) .addStringOption(option => option .setName("preco") .setDescription("Preço do produto") .setRequired(true) ) .addIntegerOption(option => option .setName("quantidade") .setDescription("Quantidade disponível") .setRequired(true) .setMinValue(1) ), new SlashCommandBuilder() .setName("painel") .setDescription("Envia o painel de compra") ].map(command => command.toJSON()); // =============================== // REGISTRAR COMANDOS // =============================== async function registrarComandos() { const rest = new REST({ version: "10" }) .setToken(process.env.DISCORD_TOKEN); await rest.put( Routes.applicationGuildCommands( process.env.CLIENT_ID, process.env.GUILD_ID ), { body: commands } ); console.log("✅ Comandos registrados!"); } // =============================== // BOT ONLINE // =============================== client.once("ready", () => { console.log( `🪽 SERAPHIM BOT online como ${client.user.tag}` ); }); // =============================== // INTERAÇÕES // =============================== client.on("interactionCreate", async interaction => { // ------------------------------- // SLASH COMMANDS // ------------------------------- if (interaction.isChatInputCommand()) { // /configurar if (interaction.commandName === "configurar") { produto.nome = interaction.options.getString("nome"); produto.preco = interaction.options.getString("preco"); produto.estoque = interaction.options.getInteger("quantidade"); return interaction.reply({ content: `✅ **Produto atualizado!**\n\n` + `📦 ${produto.nome}\n` + `💰 R$ ${produto.preco}\n` + `📦 Estoque: ${produto.estoque}`, ephemeral: true }); } // /painel if (interaction.commandName === "painel") { const embed = new EmbedBuilder() .setTitle("🪽 SERAPHIM HUB") .setDescription( `📦 **${produto.nome}**\n\n` + `💰 **R$ ${produto.preco}**\n` + `📦 **Estoque:** ${produto.estoque}\n\n` + `Clique no botão abaixo para comprar.` ); const botoes = new ActionRowBuilder() .addComponents( new ButtonBuilder() .setCustomId("comprar") .setLabel("Comprar") .setEmoji("🛒") .setStyle(ButtonStyle.Success) ); return interaction.reply({ embeds: [embed], components: [botoes] }); } } // ------------------------------- // BOTÕES // ------------------------------- if (!interaction.isButton()) return; // COMPRAR if (interaction.customId === "comprar") { if (produto.estoque <= 0) { return interaction.reply({ content: "❌ Produto sem estoque.", ephemeral: true }); } const pedido = new EmbedBuilder() .setTitle("🧾 Pedido criado") .setDescription( `📦 **Produto:** ${produto.nome}\n` + `🔢 **Quantidade:** 1\n` + `💰 **Total:** R$ ${produto.preco}\n\n` + `💳 Faça o pagamento pelo Pix configurado pelo administrador.\n\n` + `📸 Depois do pagamento, clique em ` + `**Enviar comprovante**.` ); const botoes = new ActionRowBuilder() .addComponents( new ButtonBuilder() .setCustomId("enviar_comprovante") .setLabel("Enviar comprovante") .setEmoji("📸") .setStyle(ButtonStyle.Primary) ); return interaction.reply({ embeds: [pedido], components: [botoes], ephemeral: true }); } // COMPROVANTE if (interaction.customId === "enviar_comprovante") { const guild = interaction.guild; const canal = await guild.channels.create({ name: `pedido-${interaction.user.id}`, type: ChannelType.GuildText, permissionOverwrites: [ { id: guild.roles.everyone.id, deny: [ PermissionFlagsBits.ViewChannel ] }, { id: interaction.user.id, allow: [ PermissionFlagsBits.ViewChannel, PermissionFlagsBits.SendMessages, PermissionFlagsBits.AttachFiles ] } ] }); await canal.send({ content: `🎫 **Pedido de ${interaction.user}**\n\n` + `📦 Produto: **${produto.nome}**\n` + `💰 Valor: **R$ ${produto.preco}**\n\n` + `📸 **Envie o comprovante aqui.**\n\n` + `⏳ Aguardando análise do administrador.` }); return interaction.reply({ content: `✅ Seu ticket foi criado: ${canal}`, ephemeral: true }); } }); // =============================== // INICIAR // =============================== async function iniciar() { try { if (!process.env.DISCORD_TOKEN) { throw new Error( "DISCORD_TOKEN não configurado." ); } if (!process.env.CLIENT_ID) { throw new Error( "CLIENT_ID não configurado." ); } if (!process.env.GUILD_ID) { throw new Error( "GUILD_ID não configurado." ); } await registrarComandos(); await client.login( process.env.DISCORD_TOKEN ); } catch (erro) { console.error( "❌ Erro ao iniciar:", erro ); } } iniciar();
